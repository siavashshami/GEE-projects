import pandas as pd
import numpy as np
import os
import glob

# Folder path
folder_path = r"C:\Users\siavashshami\Documents\test"

print("Scanning directory for paired files...")

# Find all CSV files and extract region codes
land_cover_files = glob.glob(os.path.join(folder_path, "*-land-cover.csv"))
regions = []
for lc_file in land_cover_files:
    region_code = os.path.basename(lc_file).split('-')[0]
    se_file = os.path.join(folder_path, f"{region_code}-soil-erosion(tonnes-ha-year).csv")
    if os.path.exists(se_file):
        regions.append(region_code)

print(f"Processing {len(regions)} regions: {', '.join(regions)}")

# Sort regions alphabetically
regions.sort()

if len(regions) == 0:
    print("No matching CSV file pairs found!")
    print("Looking for files like: region-land-cover.csv + region-soil-erosion(tonnes-ha-year).csv")
    exit()

# Land cover classes
classes = ['EvergreenForest', 'DeciduousForest', 'MixedForest', 'Shrublands', 
           'Savanna', 'Grasslands', 'Wetlands', 'Agriculture', 'Urban', 
           'SnowIce', 'Barren', 'Water']

# Classes to always filter out
always_filter_classes = ['Wetlands', 'Urban', 'SnowIce', 'Water']

def clean_numeric_columns(df):
    """Remove commas from all numeric columns and convert to float"""
    numeric_columns = df.select_dtypes(include=['object']).columns.drop('year', errors='ignore')
    for col in numeric_columns:
        df[col] = df[col].astype(str).str.replace(',', '').astype(float)
    if 'year' in df.columns:
        df['year'] = df['year'].astype(str).str.replace(',', '').astype(float).round(0).astype(int)
    return df

# Load all data + determine valid classes (>1000 km², exclude always_filter)
data_pairs = {}
valid_classes_by_region = {}
all_summary_stats = {}

for region in regions:
    lc_file = os.path.join(folder_path, f"{region}-land-cover.csv")
    se_file = os.path.join(folder_path, f"{region}-soil-erosion(tonnes-ha-year).csv")
    
    lc_df = clean_numeric_columns(pd.read_csv(lc_file))
    se_df = clean_numeric_columns(pd.read_csv(se_file))
    
    data_pairs[region] = {'landcover': lc_df, 'soil_erosion': se_df}
    
    # Valid classes: >1000 km² in any year AND not always filtered
    valid_classes = []
    for class_name in classes:
        if class_name in always_filter_classes:
            continue
        if class_name in lc_df.columns:
            if (lc_df[class_name] > 1000).any():
                valid_classes.append(class_name)
    
    valid_classes_by_region[region] = valid_classes
    print(f"{region}: Valid classes: {valid_classes}")

# === COMPREHENSIVE STATISTICS CALCULATION ===
output_content = []
output_content.append("=== SOIL EROSION & LAND COVER ANALYSIS ===")
output_content.append(f"Date: {pd.Timestamp.now().strftime('%Y-%m-%d %H:%M')}")
output_content.append(f"Regions analyzed: {', '.join(regions)}")
output_content.append(f"Classes filtered: {always_filter_classes} + <1000 km²")
output_content.append("")

for region in regions:
    lc_data = data_pairs[region]['landcover']
    se_data = data_pairs[region]['soil_erosion']
    years = lc_data['year'].values
    valid_classes = valid_classes_by_region[region]
    
    # 1. MEANS across all years
    lc_mean = lc_data[classes].mean(numeric_only=True).reindex(valid_classes).dropna()
    se_mean = se_data[classes].mean(numeric_only=True).reindex(valid_classes).dropna()
    total_area = lc_mean.sum()
    
    # 2. TOTAL EROSION (ton/year) = Area_km2 * Rate_ton/ha/yr * 100_ha/km2
    total_erosion_ton_year = lc_mean * se_mean * 100
    
    # 3. ANNUAL CHANGE RATE (%/year)
    n_years = len(years)
    annual_change = {}
    for cls in valid_classes:
        start = lc_data[cls].iloc[0]
        end = lc_data[cls].iloc[-1]
        if start > 0:
            rate = ((end - start) / start / (n_years - 1)) * 100
            annual_change[cls] = rate
    
    # 4. RISK INDEX (ton/year per km² region)
    risk_index = total_erosion_ton_year / total_area
    
    # === PURE NUMERICAL STABILITY ANALYSIS (NO THRESHOLDS) ===
    stability_analysis = {}
    
    output_content.append(f"\n{'='*70}")
    output_content.append(f"REGION: {region.upper()}")
    output_content.append(f"Years: {years.min()}-{years.max()} ({len(years)} years)")
    output_content.append(f"Total Area: {total_area:.0f} km²")
    output_content.append(f"Valid Classes ({len(valid_classes)}): {', '.join(valid_classes)}")
    output_content.append("")
    
    # Stability Analysis Table - Pure Numbers
    output_content.append(f"{'='*70}")
    output_content.append("CLASS STABILITY ANALYSIS")
    output_content.append("CV = Std/Mean * 100 | Lower = More Stable")
    output_content.append("CLASS           | CV_Area% | CV_Erosion% | CV_Combined%")
    output_content.append("-" * 65)
    
    for cls in valid_classes:
        if cls in lc_data.columns and cls in se_data.columns:
            # Coefficient of Variation for Area (Land Cover)
            cv_area = (lc_data[cls].std() / lc_data[cls].mean()) * 100 if lc_data[cls].mean() > 0 else np.nan
            
            # Coefficient of Variation for Erosion Rate  
            cv_erosion = (se_data[cls].std() / se_data[cls].mean()) * 100 if se_data[cls].mean() > 0 else np.nan
            
            # Combined Stability Score (average of both CVs)
            cv_combined = np.nanmean([cv_area, cv_erosion])
            
            stability_analysis[cls] = {
                'cv_area': cv_area,
                'cv_erosion': cv_erosion, 
                'cv_combined': cv_combined
            }
            
            # Pure numerical output - no labels
            output_content.append(f"{cls:15} | {cv_area:7.1f}% | {cv_erosion:9.1f}% | {cv_combined:10.1f}%")
    
    # Numerical Ranking (Lowest ? Highest CV)
    stability_ranking = sorted(stability_analysis.items(), 
                              key=lambda x: x[1]['cv_combined'])
    
    output_content.append("")
    output_content.append("STABILITY RANKING (Lowest ? Highest CV):")
    for i, (cls, stats) in enumerate(stability_ranking, 1):
        output_content.append(f"{i:2d}. {cls:<15}: CV={stats['cv_combined']:6.1f}%")
    
    # Show top 3 highest CV (most variable)
    output_content.append("")
    output_content.append("HIGHEST CV (Most Variable):")
    for i, (cls, stats) in enumerate(reversed(stability_ranking[-3:]), 1):
        output_content.append(f"{i}. {cls}: CV={stats['cv_combined']:6.1f}%")
    
    # Summary Statistics (NO CORRELATION)
    output_content.append(f"\n{'='*70}")
    output_content.append("SUMMARY STATISTICS")
    
    # Dominant Cover
    top_cover = lc_mean.idxmax()
    top_cover_pct = (lc_mean[top_cover] / total_area) * 100
    output_content.append(f"Top Land Cover: {top_cover}: {lc_mean[top_cover]:.0f} km² ({top_cover_pct:.1f}%)")
    
    # Top Erosion Contributor
    top_erosion = total_erosion_ton_year.idxmax()
    top_erosion_pct = (total_erosion_ton_year[top_erosion] / total_erosion_ton_year.sum()) * 100
    output_content.append(f"Top Erosion: {top_erosion}: {total_erosion_ton_year[top_erosion]:.0f} ton/year ({top_erosion_pct:.1f}%)")
    
    # Risk Index only
    output_content.append(f"Region Risk Index: {risk_index.mean():.2f} ton/year/km²")
    
    # Detailed Erosion Table
    output_content.append("\nCLASS            | Area_km2  | Rate_t/ha/yr | Total_ton/yr | %Total  | Ann.Change%/yr")
    output_content.append("-" * 85)
    for cls in valid_classes:
        area = lc_mean[cls]
        rate = se_mean[cls]
        total_ton = total_erosion_ton_year[cls]
        pct_total = (total_ton / total_erosion_ton_year.sum()) * 100
        change = annual_change.get(cls, np.nan)
        output_content.append(f"{cls:15} | {area:9.0f} | {rate:11.2f} | {total_ton:11.0f} | {pct_total:7.1f}% | {change:+7.2f}%")
    
    output_content.append(f"\nTOTAL EROSION: {total_erosion_ton_year.sum():.0f} ton/year")
    
    # Store for cross-region summary
    all_summary_stats[region] = {
        'total_area_km2': total_area,
        'lc_mean': lc_mean,
        'se_mean': se_mean,
        'total_erosion_ton_year': total_erosion_ton_year,
        'annual_change_pct_year': annual_change,
        'risk_index': risk_index,
        'stability_analysis': stability_analysis
    }

# === CROSS-REGION SUMMARY ===
output_content.append(f"\n{'='*70}")
output_content.append("CROSS-REGION COMPARISON")
output_content.append("-" * 70)

# Total erosion ranking
region_totals = {r: all_summary_stats[r]['total_erosion_ton_year'].sum() for r in regions}
top_region = max(region_totals, key=region_totals.get)
output_content.append(f"Highest Erosion Region: {top_region}: {region_totals[top_region]:.0f} ton/year")

# Average risk index
avg_risk = np.mean([all_summary_stats[r]['risk_index'].mean() for r in regions])
output_content.append(f"Average Risk Index: {avg_risk:.2f} ton/year/km²")

# Cross-region stability summary - highest CV per region
output_content.append("\nHIGHEST CV CLASSES BY REGION:")
for region in regions:
    highest_cv_class = max(all_summary_stats[region]['stability_analysis'].items(), 
                          key=lambda x: x[1]['cv_combined'])
    output_content.append(f"{region}: {highest_cv_class[0]} (CV={highest_cv_class[1]['cv_combined']:.1f}%)")

# Save comprehensive report
output_file = os.path.join(folder_path, "complete_numerical_stability_analysis.txt")
with open(output_file, 'w', encoding='utf-8') as f:
    f.write('\n'.join(output_content))

print(f"\nCOMPLETE NUMERICAL ANALYSIS SAVED: {output_file}")
print("\nNUMERICAL SUMMARY:")
for region in regions:
    total_ton = all_summary_stats[region]['total_erosion_ton_year'].sum()
    most_variable = max(all_summary_stats[region]['stability_analysis'].items(), 
                       key=lambda x: x[1]['cv_combined'])
    print(f"{region}: {total_ton:.0f} ton/year | Most Variable: {most_variable[0]} (CV={most_variable[1]['cv_combined']:.1f}%)")
