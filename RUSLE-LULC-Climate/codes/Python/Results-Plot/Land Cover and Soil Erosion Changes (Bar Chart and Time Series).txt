import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import os
import glob
from matplotlib.gridspec import GridSpec
from matplotlib.ticker import ScalarFormatter



plt.rcParams['font.family'] = ['Times New Roman']
plt.rcParams['axes.unicode_minus'] = False


plt.rcParams['figure.dpi'] = 600
plt.rcParams['savefig.dpi'] = 600  


# Disable scientific notation globally
plt.rcParams['axes.formatter.useoffset'] = False
plt.rcParams['axes.formatter.limits'] = [-5, 5]



# Folder path
folder_path = r"C:\Users\siavashshami\Documents\test"



print("Scanning directory for paired files...")



# Find all CSV files and extract region codes
land_cover_files = glob.glob(os.path.join(folder_path, "*-land-cover.csv"))
regions = []
for lc_file in land_cover_files:
    region_code = os.path.basename(lc_file).split('-')[0]
    se_file = os.path.join(folder_path, f"{region_code}-soil-erosion(tonnes-ha-year).csv")
    if os.path.exists(se_file):
        regions.append(region_code)



print(f"Processing {len(regions)} regions: {', '.join(regions)}")



# Sort regions alphabetically
regions.sort()



if len(regions) == 0:
    print("No matching CSV file pairs found!")
    print("Looking for files like: region-land-cover.csv + region-soil-erosion(tonnes-ha-year).csv")
    exit()



# Land cover classes
classes = ['EvergreenForest', 'DeciduousForest', 'MixedForest', 'Shrublands', 
           'Savanna', 'Grasslands', 'Wetlands', 'Agriculture', 'Urban', 
           'SnowIce', 'Barren', 'Water']



# Classes to always filter out
always_filter_classes = ['Wetlands', 'Urban', 'SnowIce', 'Water']



# CUSTOM COLORS
custom_colors = ['#05450a', '#78d203', '#009900', '#c6b044', '#fbff13', 
                 '#b6ff05', '#27ff87', '#ff6d4c', '#a5a5a5', '#69fff8', 
                 '#f9ffa4', '#1c0dff']



def clean_numeric_columns(df):
    """Remove commas from all numeric columns and convert to float"""
    numeric_columns = df.select_dtypes(include=['object']).columns.drop('year', errors='ignore')
    for col in numeric_columns:
        df[col] = df[col].astype(str).str.replace(',', '').astype(float)
    if 'year' in df.columns:
        # ROUND YEARS TO NEAREST INTEGER (e.g., 2004.5 -> 2004)
        df['year'] = df['year'].astype(str).str.replace(',', '').astype(float).round(0).astype(int)
    return df



# Load all data
data_pairs = {}
valid_classes_by_region = {}  # Store valid classes for each region/year


for region in regions:
    lc_file = os.path.join(folder_path, f"{region}-land-cover.csv")
    se_file = os.path.join(folder_path, f"{region}-soil-erosion(tonnes-ha-year).csv")
    
    lc_df = clean_numeric_columns(pd.read_csv(lc_file))
    se_df = clean_numeric_columns(pd.read_csv(se_file))
    
    data_pairs[region] = {'landcover': lc_df, 'soil_erosion': se_df}
    
    # Determine valid classes for this region (classes with >1000 km² in any year and not always filtered)
    valid_classes = []
    for class_name in classes:
        if class_name in always_filter_classes:
            continue
        if class_name in lc_df.columns:
            # Check if any year has >1000 km²
            if (lc_df[class_name] > 1000).any():
                valid_classes.append(class_name)
    
    valid_classes_by_region[region] = valid_classes
    print(f"{region}: Valid classes for bar chart: {valid_classes}")



# Create figure - INCREASED bottom margin for legend
fig = plt.figure(figsize=(10, 6), facecolor='white')



# GridSpec - bottom=0.22 for legend space (increased from 0.18)
gs_main = GridSpec(3, 2, 
                   width_ratios=[1, 1],
                   height_ratios=[1, 1, 1],
                   hspace=0.35, wspace=0.25,
                   left=0.06, right=0.96, bottom=0.12, top=0.96)



# Split regions
regions_col1 = regions[:3]
regions_col2 = regions[3:6]
all_regions = regions_col1 + regions_col2



for idx, region in enumerate(all_regions):
    # Calculate subplot position
    if region in regions_col1:
        row = regions_col1.index(region)
        col = 0
    else:
        row = regions_col2.index(region)
        col = 1
    
    ax = fig.add_subplot(gs_main[row, col])
    
    # Get data - EXACT CSV VALUES
    lc_data = data_pairs[region]['landcover']
    se_data = data_pairs[region]['soil_erosion']
    years = lc_data['year'].values
    valid_classes = valid_classes_by_region[region]
    
    # 1. LAND COVER BAR CHART (FILTERED: only valid classes >1000 km², exclude always_filter_classes)
    bar_width = 0.8 / len(valid_classes) if len(valid_classes) > 0 else 0.8  # avoid division by zero  # <<<
    
    for i, class_name in enumerate(valid_classes):
        # USE INTEGER YEARS FOR POSITIONING (remove .5 decimals)
        x_pos = [int(y) + i * bar_width for y in years]
        areas = lc_data[class_name].values
        color_idx = classes.index(class_name)
        ax.bar(x_pos, areas, 
               width=bar_width, 
               color=custom_colors[color_idx], 
               alpha=0.8, 
               edgecolor='black', linewidth=0.4)
    
    # 2. SOIL EROSION TIME SERIES (ONLY for classes shown in bar chart)
    ax2 = ax.twinx()
    for class_name in valid_classes:
        color_idx = classes.index(class_name)
        if class_name in se_data.columns:
            # USE INTEGER YEARS FOR PLOTTING (remove .5 decimals)
            plot_years = [int(y) for y in years]
            erosion_data = se_data[class_name].values
            ax2.plot(plot_years, erosion_data, 
                     color=custom_colors[color_idx], linewidth=1.2, 
                     marker='o', markersize=3.5, alpha=0.9)
    
    # X-axis - ONLY SHOW INTEGER YEARS, one yes one no
    all_years_int = [int(y) for y in years]  # <<<
    # keep every other year (e.g. even years); change condition to y % 2 != 0 for odd years  # <<<
    xticks_filtered = [y for y in all_years_int if y % 2 == 1]  # <<<  
    ax.set_xticks(xticks_filtered)  # <<<
    ax.tick_params(axis='x', labelsize=10, colors='black')
    
    # Y-axis labels - ONLY SPECIFIC POSITIONS
    # Area label: ONLY column 0, row 1 (middle left)
    if col == 0 and row == 1:
        ax.set_ylabel('LULC Area (Km²)', fontsize=12, fontweight='bold', color='black')
    else:
        ax.set_ylabel('')
    
    # Erosion label: ONLY column 1, row 1 (middle right)  
    if col == 1 and row == 1:
        ax2.set_ylabel('Mean Soil Erosion Rate (t/ha/yr)', fontsize=12, fontweight='bold', color='black')
    else:
        ax2.set_ylabel('')
    
    # Y-axis formatting (all subplots)
    ax.tick_params(axis='y', labelcolor='black', labelsize=10)
    ax.yaxis.set_major_formatter(ScalarFormatter(useOffset=False))
    ax.ticklabel_format(style='plain', axis='y', useOffset=False)
    
    ax2.tick_params(axis='y', labelcolor='black', labelsize=10)
    ax2.yaxis.set_major_formatter(ScalarFormatter(useOffset=False))
    ax2.ticklabel_format(style='plain', axis='y', useOffset=False)
    
    # Region name (top-right)
    ax.text(0.98, 0.98, region, transform=ax.transAxes, 
            fontsize=14, fontweight='bold', va='top', ha='right', color='black',
            bbox=dict(boxstyle='round,pad=0.3', facecolor='white', alpha=0.9,
                      edgecolor='black', linewidth=1))
    
    # Clean white background
    ax.grid(False)
    ax.set_facecolor('white')
    ax2.set_facecolor('white')
    
    # Black spines only
    for spine in ax.spines.values():
        spine.set_color('black')
    ax2.spines['right'].set_color('black')



# Legend - now fits perfectly under charts (only for classes that appear in any region)
all_used_classes = set()
for region_classes in valid_classes_by_region.values():
    all_used_classes.update(region_classes)


used_class_indices = [classes.index(cls) for cls in all_used_classes]
used_class_colors = [custom_colors[i] for i in used_class_indices]


dummy_bars = [plt.Rectangle((0,0),1,1, color=used_class_colors[i], alpha=0.8) for i in range(len(used_class_colors))]
dummy_lines = [plt.Line2D([0], [0], color=used_class_colors[i], lw=1.2, marker='o', markersize=3.5) 
               for i in range(len(used_class_colors))]


fig.legend(dummy_bars + dummy_lines, 
           [f'{list(all_used_classes)[i]}\n(area)' for i in range(len(used_class_colors))] + 
           [f'{list(all_used_classes)[i]}\n(erosion)' for i in range(len(used_class_colors))],
           loc='upper center', 
           bbox_to_anchor=(0.5, 0.07),  # Slightly higher to fit
           ncol=12, fontsize=7.5, frameon=True,  # Smaller font
           labelspacing=0.2, handlelength=1.0, columnspacing=0.6,  # Tighter spacing
           labelcolor='black', facecolor='white', edgecolor='black')



# Save outputs
output_file1 = os.path.join(folder_path, "landcover_barchart_by_year.png")
plt.savefig(output_file1, dpi=600, bbox_inches='tight', facecolor='white', 
            edgecolor='none', pad_inches=0.1)  # Reduced padding
print(f"PRESERVED: {output_file1}")



output_file2 = os.path.join(folder_path, "landcover_erosion_legend_fixed.png")
plt.savefig(output_file2, dpi=600, bbox_inches='tight', facecolor='white', 
            edgecolor='none', pad_inches=0.1)
print(f"FINAL Chart saved: {output_file2}")



plt.tight_layout()
plt.show()



# Summary
print("\nSUMMARY:")
for region in regions:
    data = data_pairs[region]
    print(f"{region}: Years={data['landcover']['year'].min()}-{data['landcover']['year'].max()}, Valid classes={valid_classes_by_region[region]}")
